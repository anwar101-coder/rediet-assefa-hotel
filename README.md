5. Supabase Database Schema Architecture

Here is the exact database schema to implement in your

Supabase SQL Editor:

SQL

-- 1. Create Rooms Table

CREATE TABLE public.rooms (

  id UUID DEFAULT

gen_random_uuid() PRIMARY KEY,

  title_en

VARCHAR(255) NOT NULL,

  title_am

VARCHAR(255),

  slug VARCHAR(255)

UNIQUE NOT NULL,

  category

VARCHAR(100) NOT NULL, -- e.g. 'Standard', 'Deluxe', 'Suite'

  price_per_night

NUMERIC(10, 2) NOT NULL,

  capacity INT DEFAULT

2,

  description_en TEXT,

  description_am TEXT,

  amenities TEXT[], --

Array of strings e.g. ['wifi', 'tv', 'balcony']

  is_available BOOLEAN

DEFAULT true,

  created_at TIMESTAMP

WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL

);

 

-- 2. Create Room Images Table

CREATE TABLE public.room_images (

  id UUID DEFAULT

gen_random_uuid() PRIMARY KEY,

  room_id UUID

REFERENCES public.rooms(id) ON DELETE CASCADE,

  image_url TEXT NOT

NULL,

  is_primary BOOLEAN

DEFAULT false,

  created_at TIMESTAMP

WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL

);

 

-- 3. Row Level Security (RLS) Rules

ALTER TABLE public.rooms ENABLE ROW LEVEL SECURITY;

 

-- Public can read active rooms

CREATE POLICY "Public rooms are viewable by

everyone" 

ON public.rooms FOR SELECT USING (true);

 

-- Only authenticated admin can insert, update, or delete

CREATE POLICY "Admins can modify rooms" 

ON public.rooms FOR ALL TO authenticated 

USING (auth.role() = 'authenticated');
